""" # ----------------------------------------- # Hilfsfunktionen # ----------------------------------------- def render_page(content, progress=0): return render_template_string( HTML, content=content, progress=progress ) def escape_html(text): text = text or "" return ( text .replace("&", "&") .replace('"', """) .replace("<", "<") .replace(">", ">") ) def clean_text(text): return (text or "").strip() def valid_color(value): if re.fullmatch(r"#[0-9A-Fa-f]{6}", value or ""): return value return "#1f2937" def wrap_text(text, font_name, font_size, max_width): words = text.split() lines = [] current_line = "" for word in words: if current_line == "": test_line = word else: test_line = current_line + " " + word width = stringWidth( test_line, font_name, font_size ) if width <= max_width: current_line = test_line else: if current_line: lines.append(current_line) current_line = word if current_line: lines.append(current_line) return lines def draw_wrapped_text( pdf, text, x, y, max_width, font_name="Helvetica", font_size=10, line_height=14 ): pdf.setFont( font_name, font_size ) for paragraph in str(text).split("\n"): lines = wrap_text( paragraph, font_name, font_size, max_width ) for line in lines: pdf.drawString( x, y, line ) y -= line_height return y # ----------------------------------------- # PDF erstellen # ----------------------------------------- def create_pdf(data, photo_path=None): buffer = BytesIO() pdf = canvas.Canvas( buffer, pagesize=A4 ) width, height = A4 margin = 50 text_color = colors.HexColor( valid_color(data.get("farbe")) ) pdf.setFillColor(text_color) pdf.setTitle( "Lebenslauf - " + data.get("vorname", "") + " " + data.get("nachname", "") ) # ------------------------------------- # Kopf # ------------------------------------- pdf.setFont( "Helvetica-Bold", 24 ) full_name = ( data.get("vorname", "") + " " + data.get("nachname", "") ) pdf.drawString( margin, height - 65, full_name ) y = height - 90 pdf.setFont( "Helvetica", 10 ) personal_info = [ data.get("adresse", ""), data.get("telefon", ""), data.get("email", ""), "Geburtsdatum: " + data.get("geburtsdatum", "") ] for info in personal_info: if info.strip(): pdf.drawString( margin, y, info ) y -= 15 # ------------------------------------- # Bewerbungsfoto # ------------------------------------- if photo_path and os.path.exists(photo_path): try: image = ImageReader(photo_path) img_width, img_height = image.getSize() box_width = 105 box_height = 130 scale = min( box_width / img_width, box_height / img_height ) final_width = img_width * scale final_height = img_height * scale image_x = width - margin - box_width image_y = height - 180 pdf.drawImage( image, image_x, image_y, width=final_width, height=final_height, preserveAspectRatio=True, mask="auto" ) except Exception: pass y -= 15 # ------------------------------------- # Überschrift # ------------------------------------- def section(title): nonlocal y if y < 100: pdf.showPage() pdf.setFillColor(text_color) y = height - 60 pdf.setFont( "Helvetica-Bold", 14 ) pdf.drawString( margin, y, title ) y -= 10 pdf.line( margin, y, width - margin, y ) y -= 20 # ------------------------------------- # Feld # ------------------------------------- def field(label, value): nonlocal y value = clean_text(value) if not value: return if y < 90: pdf.showPage() pdf.setFillColor(text_color) y = height - 60 pdf.setFont( "Helvetica-Bold", 10 ) pdf.drawString( margin, y, label ) y -= 14 y = draw_wrapped_text( pdf, value, margin, y, width - 2 * margin, font_size=10 ) y -= 8 # ------------------------------------- # Schule # ------------------------------------- section("Schulbildung") field( "Schule", data.get("schule") ) field( "Schulabschluss", data.get("abschluss") ) field( "Schulzeit", data.get("schulzeit") ) # ------------------------------------- # Erfahrung # ------------------------------------- section("Praktische Erfahrung") field( "Firma", data.get("firma") ) field( "Tätigkeit", data.get("position") ) field( "Zeitraum", data.get("zeitraum") ) field( "Aufgaben", data.get("aufgaben") ) # ------------------------------------- # Kenntnisse # ------------------------------------- section("Kenntnisse") field( "Deutsch", data.get("deutsch") ) field( "Englisch", data.get("englisch") ) field( "Computerkenntnisse", data.get("computer") ) # ------------------------------------- # Hobbys # ------------------------------------- section("Interessen") field( "Hobbys und Interessen", data.get("hobbys") ) # ------------------------------------- # Unterschrift # ------------------------------------- section("Weitere Angaben") pdf.setFont( "Helvetica", 10 ) pdf.drawString( margin, y, "Ort, Datum: ______________________________" ) y -= 35 pdf.drawString( margin, y, "Unterschrift: ______________________________" ) # ------------------------------------- # Speichern # ------------------------------------- pdf.save() buffer.seek(0) return buffer # ----------------------------------------- # Startseite # ----------------------------------------- @app.route("/") def home(): content = """

Welcome to Abdul's Lebenslauf

Lebenslauf erstellen

Python • Flask • ReportLab

""" return render_page(content) # ----------------------------------------- # Start # ----------------------------------------- @app.route("/start") def start(): return redirect( url_for( "question", index=0 ) ) # ----------------------------------------- # Fragen # ----------------------------------------- @app.route( "/question/", methods=["GET", "POST"] ) def question(index): # Ende if index >= len(QUESTIONS): return redirect( url_for("review") ) key, question_text = QUESTIONS[index] # ------------------------------------- # Daten aus Formular holen # ------------------------------------- data = {} for field_key, _ in QUESTIONS: data[field_key] = request.values.get( field_key, "" ) # ------------------------------------- # Antwort speichern # ------------------------------------- if request.method == "POST": answer = clean_text( request.form.get(key) ) data[key] = answer next_index = index + 1 # Letzte Frage if next_index >= len(QUESTIONS): return redirect_with_data( "/review", data ) return redirect_with_data( f"/question/{next_index}", data ) # ------------------------------------- # Hidden Inputs # ------------------------------------- hidden_inputs = "" for field_key, _ in QUESTIONS: if field_key == key: continue hidden_inputs += ( f'' ) # ------------------------------------- # Fortschritt # ------------------------------------- progress = int( (index / len(QUESTIONS)) * 100 ) # ------------------------------------- # Seite # ------------------------------------- content = f"""
